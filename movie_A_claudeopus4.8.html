<script>

  let 데이터 = { "1":[], "2":[], "3":[], "4":[], "5":[] };

  let 드래그중 = null;


  function 로컬저장() { localStorage.setItem("티어리스트", JSON.stringify(데이터)); }

  function 로컬불러오기() {

    const s = localStorage.getItem("티어리스트");

    if (s) { try { 데이터 = JSON.parse(s); } catch(e) {} }

  }


  function 그리기() {

    const 영역 = document.getElementById("티어영역");

    영역.innerHTML = "";

    ["1","2","3","4","5"].forEach(t => {

      const 행 = document.createElement("div");

      행.className = "티어 t" + t;

      행.innerHTML = `<div class="라벨">${t}</div>`;

      const 목록 = document.createElement("div");

      목록.className = "목록";

      목록.ondragover = e => { e.preventDefault(); 목록.classList.add("over"); };

      목록.ondragleave = () => 목록.classList.remove("over");

      목록.ondrop = e => { e.preventDefault(); 목록.classList.remove("over"); 드롭(t); };

      데이터[t].forEach((영화, i) => 목록.appendChild(카드만들기(영화, t, i)));

      행.appendChild(목록);

      영역.appendChild(행);

    });

  }


  function 카드만들기(영화, 티어, 인덱스) {

    const 카드 = document.createElement("div");

    카드.className = "카드";

    카드.draggable = true;

    const 별 = "★".repeat(영화.별점) + "☆".repeat(5 - 영화.별점);

    카드.innerHTML = `

      <button class="지우기">✕</button>

      <img src="${영화.포스터 || ''}" onerror="this.removeAttribute('src')">

      <div class="제목">${영화.제목 || '(제목없음)'}</div>

      <div class="별점">${별}</div>

    `;

    카드.querySelector(".지우기").onclick = () => {

      if (confirm("삭제할까요?")) { 데이터[티어].splice(인덱스, 1); 로컬저장(); 그리기(); }

    };

    카드.ondragstart = () => 드래그중 = { 티어, 인덱스 };

    return 카드;

  }


  function 드롭(목표티어) {

    if (!드래그중) return;

    const 영화 = 데이터[드래그중.티어].splice(드래그중.인덱스, 1)[0];

    데이터[목표티어].push(영화);

    드래그중 = null;

    로컬저장();

    그리기();

  }


  function 추가모달() { document.getElementById("모달배경").style.display = "flex"; }

  function 닫기() {

    document.getElementById("모달배경").style.display = "none";

    document.getElementById("포스터").value = "";

    document.getElementById("제목").value = "";

  }

  function 저장() {

    const 영화 = {

      포스터: document.getElementById("포스터").value,

      제목: document.getElementById("제목").value,

      별점: Number(document.getElementById("별점").value)

    };

    const t = document.getElementById("티어선택").value;

    데이터[t].push(영화);

    로컬저장();

    닫기();

    그리기();

  }


  로컬불러오기();
