<script>

  // 기본 데이터 구조

  let appData = {

    users: ["기본 사용자"],

    currentUser: "기본 사용자",

    data: {

      "기본 사용자": { "1": [], "2": [], "3": [], "4": [], "5": [] }

    }

  };


  let editMode = null; // 수정 중인 영화 정보 {tier, index}

  let dragItem = null; // 드래그 중인 영화 정보 {tier, index}


  // --- 1. 로컬 저장소 관리 ---

  function saveData() {

    localStorage.setItem("MovieTierListData", JSON.stringify(appData));

  }

  function loadData() {

    const saved = localStorage.getItem("MovieTierListData");

    if (saved) {

      try { appData = JSON.parse(saved); } catch(e) { console.error("데이터 불러오기 실패"); }

    }

  }


  // --- 2. 화면 그리기 (탭) ---

  function renderTabs() {

    const area = document.getElementById("tabsArea");

    area.innerHTML = "";

    

    appData.users.forEach(user => {

      const tab = document.createElement("div");

      tab.className = "tab" + (user === appData.currentUser ? " active" : "");

      tab.innerHTML = `<span>${user}</span>`;

      tab.onclick = () => { appData.currentUser = user; renderBoard(); renderTabs(); };

      

      if (appData.users.length > 1) {

        const delBtn = document.createElement("span");

        delBtn.className = "tab-del";

        delBtn.innerText = "✕";

        delBtn.onclick = (e) => {

          e.stopPropagation();

          if (confirm(`'${user}' 탭을 삭제하시겠습니까? 기록이 모두 지워집니다.`)) {

            appData.users = appData.users.filter(u => u !== user);

            delete appData.data[user];

            if (appData.currentUser === user) appData.currentUser = appData.users[0];

            saveData(); renderTabs(); renderBoard();

          }

        };

        tab.appendChild(delBtn);

      }

      area.appendChild(tab);

    });


    const addBtn = document.createElement("button");

    addBtn.innerText = "+ 사람 추가";

    addBtn.onclick = () => {

      const name = prompt("추가할 사람의 이름을 입력하세요:");

      if (name && !appData.users.includes(name)) {

        appData.users.push(name);

        appData.data[name] = { "1": [], "2": [], "3": [], "4": [], "5": [] };

        appData.currentUser = name;

        saveData(); renderTabs(); renderBoard();

      } else if (appData.users.includes(name)) {

        alert("이미 존재하는 이름입니다.");

      }

    };

    area.appendChild(addBtn);

  }


  // --- 3. 화면 그리기 (티어 보드) ---

  function renderBoard() {

    const area = document.getElementById("boardArea");

    area.innerHTML = "";

    const userData = appData.data[appData.currentUser];


    ["1", "2", "3", "4", "5"].forEach(tier => {

      const row = document.createElement("div");

      row.className = `tier-row tier-${tier}`;

      

      const label = document.createElement("div");

      label.className = "tier-label";

      label.innerText = tier;

      

      const list = document.createElement("div");

      list.className = "tier-list";

      

      // 드래그 앤 드롭 이벤트 설정

      list.ondragover = (e) => { e.preventDefault(); list.classList.add("drag-over"); };

      list.ondragleave = () => { list.classList.remove("drag-over"); };

      list.ondrop = (e) => {

        e.preventDefault();

        list.classList.remove("drag-over");

        if (dragItem && dragItem.tier !== tier) {

          const movie = appData.data[appData.currentUser][dragItem.tier].splice(dragItem.index, 1)[0];

          appData.data[appData.currentUser][tier].push(movie);

          saveData(); renderBoard();

        }

      };


      // 영화 카드 생성

      userData[tier].forEach((movie, index) => {

        const card = document.createElement("div");

        card.className = "movie-card";

        card.draggable = true;

        

        card.ondragstart = () => { dragItem = { tier, index }; };

        card.ondragend = () => { dragItem = null; };

        card.onclick = () => { openModal(tier, index); };


        const stars = "★".repeat(movie.rating) + "☆".repeat(5 - movie.rating);

        

        card.innerHTML = `

          <img src="${movie.poster || ''}" class="movie-poster" onerror="this.src='data:image/svg+xml;utf8,<svg xmlns=\\'http://www.w3.org/2000/svg\\' width=\\'80\\' height=\\'115\\'><rect width=\\'100%\\' height=\\'100%\\' fill=\\'%23555\\'/><text x=\\'50%\\' y=\\'50%\\' fill=\\'white\\' text-anchor=\\'middle\\' dy=\\'0.3em\\'>No Image</text></svg>'">

          <div class="movie-title">${movie.title || '제목 없음'}</div>

          <div class="movie-rating">${stars}</div>

        `;

        list.appendChild(card);

      });


      row.appendChild(label);

      row.appendChild(list);

      area.appendChild(row);

    });

  }


  // --- 4. 모달 (추가/수정) 관리 ---

  function openModal(tier = null, index = null) {

    editMode = tier !== null ? { tier, index } : null;

    document.getElementById("modalTitle").innerText = editMode ? "영화 수정" : "영화 추가";

    document.getElementById("btnDelete").style.display = editMode ? "block" : "none";

    

    if (editMode) {

      const movie = appData.data[appData.currentUser][tier][index];

      document.getElementById("mTier").value = tier;

      document.getElementById("mPoster").value = movie.poster || "";

      document.getElementById("mTitle").value = movie.title || "";

      document.getElementById("mYear").value = movie.year || "";

      document.getElementById("mRating").value = movie.rating || "5";

      document.getElementById("mDate").value = movie.date || "";

      document.getElementById("mReview").value = movie.review || "";

      document.getElementById("mWatcher").value = movie.watcher || appData.currentUser;

    } else {

      document.getElementById("mTier").value = "1";

      document.getElementById("mPoster").value = "";

      document.getElementById("mTitle").value = "";

      document.getElementById("mYear").value = "";

      document.getElementById("mRating").value = "5";

      document.getElementById("mDate").value = new Date().toISOString().split('T')[0];

      document.getElementById("mReview").value = "";

      document.getElementById("mWatcher").value = appData.currentUser;

    }

    document.getElementById("modalOverlay").style.display = "flex";

  }


  function closeModal() {

    document.getElementById("modalOverlay").style.display = "none";

  }


  function saveMovie() {

    const targetTier = document.getElementById("mTier").value;

    const movieData = {

      poster: document.getElementById("mPoster").value,

      title: document.getElementById("mTitle").value,

      year: document.getElementById("mYear").value,

      rating: Number(document.getElementById("mRating").value),

      date: document.getElementById("mDate").value,

      review: document.getElementById("mReview").value,

      watcher: document.getElementById("mWatcher").value

    };


    if (editMode) {

      // 기존 위치에서 삭제

      appData.data[appData.currentUser][editMode.tier].splice(editMode.index, 1);

    }

    // 새 위치(또는 기존 위치)에 추가

    appData.data[appData.currentUser][targetTier].push(movieData);

    

    saveData(); renderBoard(); closeModal();

  }


  function deleteMovie() {

    if (confirm("이 영화를 삭제하시겠습니까?")) {

      appData.data[appData.currentUser][editMode.tier].splice(editMode.index, 1);

      saveData(); renderBoard(); closeModal();

    }

  }


  // --- 5. JSON 내보내기 / 불러오기 ---

  function exportJSON() {

    const dataStr = JSON.stringify(appData, null, 2);

    const blob = new Blob([dataStr], { type: "application/json" });

    const url = URL.createObjectURL(blob);

    const a = document.createElement("a");

    a.href = url;

    a.download = "영화티어리스트_백업.json";

    a.click();

    URL.revokeObjectURL(url);

  }


  function importJSON(event) {

    const file = event.target.files[0];

    if (!file) return;

    const reader = new FileReader();

    reader.onload = (e) => {

      try {

        const importedData = JSON.parse(e.target.result);

        if (importedData.users && importedData.data) {

          appData = importedData;

          saveData(); renderTabs(); renderBoard();

          alert("데이터를 성공적으로 불러왔습니다!");

        } else {

          alert("올바른 티어리스트 파일이 아닙니다.");

        }

      } catch (err) {

        alert("파일을 읽는 중 오류가 발생했습니다.");

      }

    };

    reader.readAsText(file);

    event.target.value = ""; // 초기화

  }


  // --- 초기 실행 ---

  loadData();

  renderTabs();

  renderBoard();

</script>
